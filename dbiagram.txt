Table Rtu{
  id integer [primary key]
  title varchar
  code varchar
}

Table CustHouse{
  id integer [primary key]
  title varchar
  code varchar
  upper_id integer [not null, ref: > Rtu.id]
}

Table CustPost{
  id integer [primary key]
  title varchar
  code varchar
  upper_id integer [not null, ref: > CustHouse.id]
}

Table Ppr{
  id integer [primary key]
  pptype varchar
  title varchar
  tow_country varchar
}

Table Mmpo{
  id integer [primary key]
  title varchar
}

Table Oez{
  id integer [primary key]
  title varchar
}


Table Ztk{
  id integer [primary key]
  title varchar
}

Table LocationOfUse{
  id integer [primary key]
  ppr integer [null, unique, ref: - Ppr.id]
  mmpo integer [null, unique, ref: - Mmpo.id]
  oez integer [null, unique, ref: - Oez.id]
  ztk integer [null, unique, ref: - Ztk.id]
}

Table CustPlace1Acc{
  id integer [primary key]
  rtu integer [null, unique, ref: - Rtu.id]
  custhouse integer [null, unique, ref: - CustHouse.id]
  custpost integer [null, unique, ref: - CustPost.id]
}

Table CustPlace1Use{
  id integer [primary key]
  rtu integer [null, unique, ref: - Rtu.id]
  custhouse integer [null, unique, ref: - CustHouse.id]
  custpost integer [null, unique, ref: - CustPost.id]

Table CustPlace2{

  title varchar
  code varchar
  level varchar
  upper_id integer [not null, ref: > CustPlace2.id]

Table SourceTypes{
