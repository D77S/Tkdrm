Table Rtu{
  id integer [primary key]
  title varchar
  code varchar
  address varchar
  ztk_allowed boolean
  standalone_allowed boolean
}

Table CustHouse{
  id integer [primary key]
  title varchar
  code varchar
  address varchar
  ztk_allowed boolean
  standalone_allowed boolean
  upper_id integer [not null, ref: > Rtu.id]
}

Table CustPost{
  id integer [primary key]
  title varchar
  code varchar
  address varchar
  ztk_allowed boolean
  standalone_allowed boolean
  upper_id integer [not null, ref: > CustHouse.id]
}

Table Ppr{

Table Oez{

Table Ztk{

Table CustPlace1Acc{

  ppr integer [null, unique, ref: - Ppr.id]

  oez integer [null, unique, ref: - Oez.id]
  ztk integer [null, unique, ref: - Ztk.id]

Table DevTypes{
  id integer [primary key]
  title varchar

  serial_flag varchar
  upper_dev_flag boolean
  si_flag boolean

Table Device{
  id integer [primary key]
  type integer [not null, ref: > DevTypes.id]

  cp2_acc integer [not null, ref: > CustPlace2.id]
