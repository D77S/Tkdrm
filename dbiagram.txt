Table Rtu{
  id integer [primary key]
  title varchar
  code varchar
  ztk_allowed boolean
  standalone_allowed boolean
}

Table CustHouse{
  id integer [primary key]
  title varchar
  code varchar
  ztk_allowed boolean
  standalone_allowed boolean
  upper_id integer [not null, ref: > Rtu.id]
}

Table CustPost{
  id integer [primary key]
  title varchar
  code varchar
  ztk_allowed boolean
  standalone_allowed boolean
  upper_id integer [not null, ref: > CustHouse.id]
}

Table Ppr{
  id integer [primary key]
  pptype varchar
  title varchar
  tow_country varchar
}

Table Mmpo{
  id integer [primary key]
  title varchar
}

Table Oez{
  id integer [primary key]
  title varchar
}


Table Ztk{
  id integer [primary key]
  title varchar
}

Table CustPlace1Acc{
  id integer [primary key]
  rtu integer [null, unique, ref: - Rtu.id]
  custhouse integer [null, unique, ref: - CustHouse.id]
  custpost integer [null, unique, ref: - CustPost.id]
  ztk_allowed boolean
  stanalone_allowed boolean
}

Table CustPlace1Use{
  id integer [primary key]
  rtu integer [null, unique, ref: - Rtu.id]
  custhouse integer [null, unique, ref: - CustHouse.id]
  custpost integer [null, unique, ref: - CustPost.id]
  ztk_allowed boolean
  stanalone_allowed boolean
}

Table CustPlace2{
  id integer [primary key]
  title varchar
  code varchar
  level varchar
  upper_id integer [not null, ref: > CustPlace2.id]
  ztk_allowed boolean
  stanalone_allowed boolean
}

Table SourceTypes{
  id integer [primary key]
  title varchar
}

Table LocationOfUse{
  id integer [primary key]
  ppr integer [null, unique, ref: - Ppr.id]
  mmpo integer [null, unique, ref: - Mmpo.id]
  oez integer [null, unique, ref: - Oez.id]
  ztk integer [null, unique, ref: - Ztk.id]
  is_ztk boolean
}

Table CustPlaceToLocation{
  id integer [primary key]
  cust_pl1 integer [not null, ref: > CustPlace1Use.id]
  cust_pl2 integer [not null, ref: > CustPlace2.id]
  loc integer [not null, ref: > LocationOfUse.id]
  is_main boolean
}

Table DevCats{
  id integer [primary key]
  title varchar
}

Table DevTypes{
  id integer [primary key]
  title varchar
  category integer [not null, ref: > DevCats.id]
  serial_flag varchar
  sub_types JSON
}

Table Device{
  id integer [primary key]
  type integer [not null, ref: > DevTypes.id]
  serial varchar
  cp1_acc integer [not null, ref: > CustPlace1Acc.id]
  cp2_acc integer [not null, ref: > CustPlace2.id]
  sour_type integer [not null, ref: > SourceTypes.id]
  sub_type varchar
  // rel_of_work integer [not null, ref: <> CustPlaceToLocation.id]
}

Table RelToDev{
  id integer [primary key]
  to_rel integer [not null, ref: > CustPlaceToLocation.id]
  to_dev integer [not null, ref: > Device.id]
  is_main boolean
}
